import boto3
from dataclasses import dataclass
from typing import List

@dataclass
class S3Bucket:
    name: str
    public_accessible: bool
    objects_count: int
    size_bytes: float
    versioning: bool
    encryption: bool
    risk_level: str

@dataclass
class EC2Instance:
    instance_id: str
    ip_address: str
    security_group: str
    open_ports: List[int]
    public_facing: bool
    risk_level: str

class AWSEnumerator:
    def __init__(self, access_key: str, secret_key: str, region: str = 'us-east-1'):
        self.s3_client = boto3.client('s3', 
                                      aws_access_key_id=access_key,
                                      aws_secret_access_key=secret_key)
        self.ec2_client = boto3.client('ec2', region_name=region)
        self.iam_client = boto3.client('iam', 
                                       aws_access_key_id=access_key,
                                       aws_secret_access_key=secret_key)
    
    def enumerate_s3_buckets(self) -> List[S3Bucket]:
        """
        List all S3 buckets in account
        Check: public ACL, encryption, versioning
        Return: S3Bucket objects with risk assessment
        """
        buckets = []
        response = self.s3_client.list_buckets()
        
        for bucket in response['Buckets']:
            bucket_name = bucket['Name']
            
            # Check if public
            try:
                acl = self.s3_client.get_bucket_acl(Bucket=bucket_name)
                is_public = any(g['Grantee'].get('URI') == 'http://acs.amazonaws.com/groups/global/AllUsers'
                               for g in acl['Grants'])
            except:
                is_public = False
            
            # Calculate size
            size = self._get_bucket_size(bucket_name)
            
            # Risk assessment
            risk = 'CRITICAL' if is_public else ('HIGH' if not size else 'MEDIUM')
            
            buckets.append(S3Bucket(
                name=bucket_name,
                public_accessible=is_public,
                objects_count=self._count_objects(bucket_name),
                size_bytes=size,
                encryption=self._check_encryption(bucket_name),
                versioning=self._check_versioning(bucket_name),
                risk_level=risk
            ))
        
        return buckets
    
    def enumerate_ec2_instances(self) -> List[EC2Instance]:
        """
        List all EC2 instances
        Check: security groups, public IPs, ports
        Return: EC2Instance objects with risk assessment
        """
        instances = []
        response = self.ec2_client.describe_instances()
        
        for reservation in response['Reservations']:
            for instance in reservation['Instances']:
                # Get open ports from security group
                sg = instance['SecurityGroups'][0]
                open_ports = self._get_open_ports_from_sg(sg['GroupId'])
                
                # Assess risk
                is_public = instance.get('PublicIpAddress') is not None
                risk = 'CRITICAL' if (is_public and 22 in open_ports) else 'HIGH' if is_public else 'MEDIUM'
                
                instances.append(EC2Instance(
                    instance_id=instance['InstanceId'],
                    ip_address=instance.get('PublicIpAddress', 'N/A'),
                    security_group=sg['GroupName'],
                    open_ports=open_ports,
                    public_facing=is_public,
                    risk_level=risk
                ))
        
        return instances
